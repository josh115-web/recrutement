Connexion Google Sheets API

Google Apps Script Web App Integration (avec Hashtags Spécialité & Skills)

Cette URL permet de synchroniser les candidats et leurs hashtags de compétences directement avec votre feuille Google Sheets en temps réel.

Code Google Apps Script (Mis à jour Hashtags) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0]) continue;
    var candidate = {
      id: row[0],
      fullName: row[1],
      email: row[2],
      specialty: row[3],
      skillsHashtags: row[4] || "",
      studyLevel: row[5],
      scheduledTime: row[6],
      arrivalTime: row[7],
      delayMinutes: Number(row[8] || 0),
      formulaireScore: Number(row[9] || 0),
      skillsScore: Number(row[10] || 0),
      rhScore: Number(row[11] || 0),
      finalScore: Number(row[12] || 0),
      remarks: row[13] || "",
      updatedAt: row[14] || new Date().toISOString()
    };
    result.push(candidate);
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var item = contents.data;
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    
    if (action === 'add') {
      sheet.appendRow([
        item.id, item.fullName, item.email, item.specialty, item.skillsHashtags, item.studyLevel,
        item.scheduledTime, item.arrivalTime, item.delayMinutes,
        item.formulaireScore, item.skillsScore, item.rhScore, item.finalScore,
        item.remarks, item.updatedAt
      ]);
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Ajouté' })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === 'update') {
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] == item.id) {
          var rowNum = i + 1;
          sheet.getRange(rowNum, 1, 1, 15).setValues([[
            item.id, item.fullName, item.email, item.specialty, item.skillsHashtags, item.studyLevel,
            item.scheduledTime, item.arrivalTime, item.delayMinutes,
            item.formulaireScore, item.skillsScore, item.rhScore, item.finalScore,
            item.remarks, item.updatedAt
          ]]);
          return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Mis à jour' })).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }
    
    if (action === 'delete') {
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] == item.id) {
          sheet.deleteRow(i + 1);
          return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Supprimé' })).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
